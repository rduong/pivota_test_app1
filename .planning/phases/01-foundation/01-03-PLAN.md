**2. Router namespace — client-side state machine:**
```javascript
const Router = {
  currentView: null,
  currentContext: {},

  navigate(view, context = {}) {
    // Role guard: admin-only views require active admin session
    const session = Storage.get(Config.KEYS.SESSION);
    if (['ADMIN_DASHBOARD', 'ADMIN_VIEW_USER'].includes(view)) {
      if (!session || session.role !== 'admin') {
        this.navigate('USER_DASHBOARD');
        return;
      }
    }
    // Guard: USER_DASHBOARD requires any valid session
    if (view === 'USER_DASHBOARD') {
      if (!session) {
        this.navigate('LOGIN');
        return;
      }
    }
    this.currentView = view;
    this.currentContext = context;
    this.render();
  },

  render() {
    const app = document.getElementById('app');
    if (!app) return;
    switch (this.currentView) {
      case 'LOGIN':
        app.innerHTML = UI.renderLogin(this.currentContext);
        break;
      case 'REGISTER':
        app.innerHTML = UI.renderRegister();
        break;
      case 'ADMIN_DASHBOARD':
        app.innerHTML = UI.renderAdminDashboard();
        break;
      case 'USER_DASHBOARD':
        app.innerHTML = UI.renderUserDashboard();
        break;
      case 'ADMIN_VIEW_USER':
        app.innerHTML = UI.renderAdminViewUser(this.currentContext);
        break;
      default:
        app.innerHTML = UI.renderLogin({});
    }
    // ALWAYS bind events AFTER innerHTML assignment — elements exist only after this point
    UI.bindEvents(this.currentView, this.currentContext);
  },

  start(firstRun = false) {
    const session = Storage.get(Config.KEYS.SESSION);
    if (!session) {
      this.navigate('LOGIN', { firstRun });
    } else if (session.role === 'admin') {
      this.navigate('ADMIN_DASHBOARD');
    } else {
      this.navigate('USER_DASHBOARD');
    }
  }
};
```

```javascript
document.addEventListener('DOMContentLoaded', async () => {
  // Guard: SubtleCrypto requires secure context (file:// or HTTPS)
  // Plain http:// will NOT work — inform the user gracefully
  if (!window.crypto || !window.crypto.subtle) {
    document.getElementById('app').innerHTML = `
      <main class="auth-page">
        <div class="auth-card" style="text-align:center">
          <p style="color:var(--color-error); font-weight:500; margin-bottom:var(--space-md)">
            ⚠️ Secure context required
          </p>
          <p style="color:var(--color-text-muted); font-size:var(--font-size-sm)">
            This app requires HTTPS or a local file (file://) to work correctly.
            Please open index.html directly from your file system.
          </p>
        </div>
      </main>
    `;
    return;
  }

  // Bootstrap: seed admin on first run, get firstRun flag
  const { firstRun } = await Bootstrap.init();

  // Start router: reads session, routes to appropriate view
  // Passes firstRun flag so LOGIN view can show first-run notice (FR-F8.3)
  Router.start(firstRun);
});
```

**Critical ordering rules:**
- UI namespace MUST be defined before Router (Router.render() calls UI methods)
- Router namespace MUST be defined before DOMContentLoaded handler (handler calls Router.start())
- DOMContentLoaded handler MUST be the last thing in the script block
- `bindEvents()` MUST be called AFTER `app.innerHTML = ...` — never before
