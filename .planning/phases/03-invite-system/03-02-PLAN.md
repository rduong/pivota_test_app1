**Part B — Wire bindEvents() for REGISTER view:**

In `UI.bindEvents()`, extend the existing `if (view === 'REGISTER')` branch to add registration form handling. The existing `go-login` handler already exists there — ADD the form handler below it:

```javascript
const registerForm = document.getElementById('register-form');
if (registerForm) {
  registerForm.addEventListener('submit', async (e) => {
    e.preventDefault();

    // Read field values (do NOT clear fields — FR-F2.4)
    const username = document.getElementById('reg-username').value;
    const password = document.getElementById('reg-password').value;
    const confirmPassword = document.getElementById('reg-confirm-password').value;
    const inviteCode = document.getElementById('reg-invite-code').value;

    // Clear all previous error messages before new submission
    ['reg-username-error', 'reg-password-error', 'reg-confirm-error', 'reg-invite-error']
      .forEach(id => { const el = document.getElementById(id); if (el) el.textContent = ''; });

    // Disable button while processing
    const btn = document.getElementById('register-btn');
    if (btn) btn.disabled = true;

    const result = await Users.register(username, password, confirmPassword, inviteCode);

    if (result.ok) {
      Router.navigate('USER_DASHBOARD');
    } else {
      // Map field name from Users.register to error span id
      const fieldToErrorId = {
        username: 'reg-username-error',
        password: 'reg-password-error',
        confirmPassword: 'reg-confirm-error',
        inviteCode: 'reg-invite-error'
      };
      const errorEl = document.getElementById(fieldToErrorId[result.field]);
      if (errorEl) errorEl.textContent = result.message;
      if (btn) btn.disabled = false;
    }
  });
}
```

**Part C — Wire bindEvents() for ADMIN_DASHBOARD view:**

Add a new `if (view === 'ADMIN_DASHBOARD')` branch to `UI.bindEvents()`:

```javascript
if (view === 'ADMIN_DASHBOARD') {
  // Generate invite code (FR-F3.1)
  const generateBtn = document.getElementById('generate-invite-btn');
  if (generateBtn) {
    generateBtn.addEventListener('click', () => {
      const invite = InviteCodes.create();
      // Show code in #new-invite-display
      const display = document.getElementById('new-invite-display');
      const codeSpan = document.getElementById('new-invite-code');
      if (display && codeSpan) {
        codeSpan.textContent = invite.code;
        display.style.display = 'flex';
        display.style.alignItems = 'center';
        display.style.marginTop = 'var(--space-md)';
      }
      // Re-render the invite list to show the new code
      // Re-render the full dashboard to show updated list
      Router.render();
    });
  }

  // Copy invite code to clipboard (FR-F3.2)
  const copyBtn = document.getElementById('copy-invite-btn');
  if (copyBtn) {
    copyBtn.addEventListener('click', async () => {
      const codeSpan = document.getElementById('new-invite-code');
      const statusEl = document.getElementById('copy-status');
      const code = codeSpan ? codeSpan.textContent : '';
      if (!code) return;
      try {
        await navigator.clipboard.writeText(code);
        if (statusEl) {
          statusEl.textContent = 'Copied!';
          setTimeout(() => { statusEl.textContent = ''; }, 2000);
        }
      } catch (err) {
        if (statusEl) statusEl.textContent = 'Could not copy — please copy the code manually.';
      }
    });
  }

  // Logout button
  const logoutBtn = document.getElementById('logout-btn');
  if (logoutBtn) {
    logoutBtn.addEventListener('click', () => {
      if (typeof Auth !== 'undefined' && Auth.logout) {
        Auth.logout();
      } else {
        Storage.remove(Config.KEYS.SESSION);
        Router.navigate('LOGIN');
      }
    });
  }
}
```

**Important:** `Router.render()` re-renders the full ADMIN_DASHBOARD — after generate-click, the re-render will call `renderAdminDashboard()` again which reads the updated `InviteCodes.list()`. The newly generated code display will be lost on re-render (acceptable — the code is in the list). To preserve the displayed code across re-render, store it in `Router.currentContext` before calling `Router.render()`:
```javascript
Router.currentContext = { ...Router.currentContext, lastGeneratedCode: invite.code };
Router.render();
// After render, re-show the display section with the last generated code
const display2 = document.getElementById('new-invite-display');
const codeSpan2 = document.getElementById('new-invite-code');
if (display2 && codeSpan2) {
  codeSpan2.textContent = invite.code;
  display2.style.display = 'flex';
  display2.style.alignItems = 'center';
  display2.style.marginTop = 'var(--space-md)';
}
```
