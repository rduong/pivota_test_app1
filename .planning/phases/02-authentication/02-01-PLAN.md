```javascript
    // ===== escapeHtml — XSS prevention for user-provided data in innerHTML =====
    function escapeHtml(str) {
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }
```

**Addition 2 — Auth namespace:**
Insert the Auth namespace AFTER the `Bootstrap` namespace closing `};` and BEFORE the `// ===== UI namespace` comment. The exact code:

```javascript
    // ===== Auth namespace — login, logout, session management (FR-F1.1–F1.5) =====
    const Auth = {
      // Validates credentials, writes session to localStorage on success.
      // Returns: { ok: true, session } | { ok: false, error: 'EMPTY_FIELDS' | 'INVALID_CREDENTIALS' }
      // MUST be awaited — calls hashPassword() which is async (SubtleCrypto)
      async login(username, password) {
        // FR-F1.2 rule 1: non-empty check
        if (!username.trim() || !password) {
          return { ok: false, error: 'EMPTY_FIELDS' };
        }

        // FR-F1.2 rule 2: user must exist
        const users = Storage.get(Config.KEYS.USERS, []);
        const user = users.find(u => u.username === username.trim());
        if (!user) {
          return { ok: false, error: 'INVALID_CREDENTIALS' };
        }

        // FR-F1.2 rule 3: hash submitted password, compare to stored hash
        // MUST await — hashPassword is async (SubtleCrypto)
        const submittedHash = await hashPassword(password);
        if (submittedHash !== user.passwordHash) {
          return { ok: false, error: 'INVALID_CREDENTIALS' };
        }

        // FR-F1.3: Write session to localStorage (exact shape per TechArch §2.2)
        const session = {
          username: user.username,
          role: user.role,          // 'admin' | 'user'
          token: generateUUID(),    // cryptographically random UUID
          loginAt: new Date().toISOString()
        };
        Storage.set(Config.KEYS.SESSION, session);

        return { ok: true, session };
      },

      // FR-F1.4: Clear session, navigate to LOGIN
      logout() {
        Storage.remove(Config.KEYS.SESSION);
        Router.navigate('LOGIN');
      },

      // FR-F1.3: Read current session (null if not logged in)
      getSession() {
        return Storage.get(Config.KEYS.SESSION, null);
      }
    };
```

**Critical ordering constraint:** Auth must appear BEFORE the UI namespace (which calls Auth in bindEvents). Auth appears AFTER Bootstrap (which only uses Storage/hashPassword/generateUUID). This mirrors the dependency chain.
